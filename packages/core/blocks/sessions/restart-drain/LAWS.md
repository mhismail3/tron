  else if bs.head.admitted.At? && (best.admitted.Unset? || bs.head.admitted.ms < best.admitted.ms) then OldestOf(bs.tail, bs.head)

```dafny
function LatestFor(bs: List<Blocker>, s: Token, latest: Time): Time {
  if bs.Nil? then latest
  else if bs.head.session.InSession? && bs.head.session.session == s && bs.head.progress.At?
    && (latest.Unset? || bs.head.progress.ms > latest.ms) then LatestFor(bs.tail, s, bs.head.progress)
  else LatestFor(bs.tail, s, latest)
}
```

```dafny
function Evaluate(count: Count, blockers: List<Blocker>, tracked: Tracked, now: Ms, limit: Ms): Step {
  if count == 0 then Step(Completed, Untracked)
  else if AllOwners(count, blockers) then Step(UnresolvedOwners, Untracked)
  else if blockers.Nil? then Step(Waiting, tracked)
  else
    var o := OldestOf(blockers.tail, blockers.head);
    var p := if o.session.NoSession? then o.progress else LatestFor(blockers, o.session.session, Unset);
    var since := if tracked.Tracking? && tracked.id == o.id && SameState(tracked.state, o.state)
      && SameTime(tracked.progress, p) && SameTime(tracked.admitted, o.admitted) then tracked.since
      else if p.At? then p.ms else if o.admitted.At? then o.admitted.ms else now;
    var age := if now <= since then 0 else now - since;
    Step(if age >= limit then Stalled(o.id, age) else Waiting, Tracking(o.id, o.state, p, o.admitted, since))
}
```

**Elems**: Elems: the items of a linked list, in order.

**Oldest**: Oldest: blocker i was admitted strictly earlier than every blocker listed before it, and no blocker listed after it was admitted strictly earlier: the first admitted at the earliest admission time, or the first blocker when none has one.

```dafny
ghost predicate Oldest(bs: seq<Blocker>, i: int) {
  0 <= i < |bs|
  && (forall j | 0 <= j < i :: Before(bs[i].admitted, bs[j].admitted))
  && (forall j | i < j < |bs| :: !Before(bs[j].admitted, bs[i].admitted))
}
```

**Before**: Before: time a is strictly earlier than time b; a time that is set is earlier than an unset one.

```dafny
ghost predicate Before(a: Time, b: Time) { a.At? && (b.Unset? || a.ms < b.ms) }
```

**Progress**: Progress: the progress time that judges blocker o: the latest progress time of its session's blockers, or its own when it has no session.

```dafny
ghost function Progress(bs: seq<Blocker>, o: Blocker): Time {
  if o.session.NoSession? then o.progress else LatestOf(bs, o.session.session)
}
```

**LatestOf**: LatestOf: the latest progress time of the blockers of session s; Unset when none has one.

```dafny
ghost function LatestOf(bs: seq<Blocker>, s: Token): Time

{ if |bs| == 0 then Unset else Later(Own(bs[0], s), LatestOf(bs[1..], s)) }
```

**Later**: Later: the later of two times; a time that is set is later than an unset one.

```dafny
ghost function Later(a: Time, b: Time): Time { if a.Unset? then b else if b.Unset? || a.ms >= b.ms then a else b }
```

**Unchanged**: Unchanged: the timer already tracks blocker o, with the same state, progress time p and admission time.

```dafny
ghost predicate Unchanged(t: Tracked, o: Blocker, p: Time) {
  t.Tracking? && t.id == o.id && t.state == o.state && t.progress == p && t.admitted == o.admitted
}
```

**Signal**: Signal: the time a blocker was last seen to move: its progress time, else its admission time, else now.

```dafny
ghost function Signal(p: Time, a: Time, now: Ms): Ms { if p.At? then p.ms else if a.At? then a.ms else now }
```

**Age**: Age: the time from since to now, never below zero.

```dafny
ghost function Age(now: Ms, since: Ms): Ms { if now <= since then 0 else now - since }
```
